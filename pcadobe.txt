# C:\ProgramData\Adobe の中身を特定する（読み取りだけ・何も消さない）
# 実行: irm https://shin3578-oss.github.io/kds-dashboard/pcadobe.txt | iex
$ErrorActionPreference = 'SilentlyContinue'
$out = New-Object System.Collections.Generic.List[string]
function W($s){ $out.Add($s); Write-Host $s }

function SizeMB($path){
  $s = (Get-ChildItem $path -Recurse -File -Force -ErrorAction SilentlyContinue |
        Measure-Object Length -Sum).Sum
  if($null -eq $s){ return 0 }
  return [math]::Round($s/1MB,0)
}

W ("=== ProgramData\Adobe の内訳  {0}  ({1}) ===" -f (Get-Date -Format 'yyyy-MM-dd HH:mm'), $env:COMPUTERNAME)
W ""

$root = 'C:\ProgramData\Adobe'
if(-not (Test-Path $root)){ W "  （フォルダがありません）"; return }

W "■ 第1階層"
Get-ChildItem $root -Directory -Force | ForEach-Object {
  [pscustomobject]@{ MB = SizeMB $_.FullName; Path = $_.FullName }
} | Sort-Object MB -Descending | ForEach-Object {
  W ("  {0,7} MB  {1}" -f $_.MB, $_.Path)
}
W ""

W "■ 第2階層（100MB以上のみ）"
Get-ChildItem $root -Directory -Force | ForEach-Object {
  Get-ChildItem $_.FullName -Directory -Force | ForEach-Object {
    $mb = SizeMB $_.FullName
    if($mb -ge 100){ [pscustomobject]@{ MB=$mb; Path=$_.FullName } }
  }
} | Sort-Object MB -Descending | Select-Object -First 25 | ForEach-Object {
  W ("  {0,7} MB  {1}" -f $_.MB, $_.Path)
}
W ""

W "■ 大きいファイル 上位20（更新日つき）"
Get-ChildItem $root -Recurse -File -Force |
  Sort-Object Length -Descending | Select-Object -First 20 | ForEach-Object {
    W ("  {0,7} MB  {1}  {2}" -f [math]::Round($_.Length/1MB,0),
        $_.LastWriteTime.ToString('yyyy-MM-dd'), $_.FullName)
  }
W ""

W "■ 拡張子ごとの合計（上位10）"
Get-ChildItem $root -Recurse -File -Force |
  Group-Object Extension |
  ForEach-Object {
    [pscustomobject]@{ Ext=$_.Name; MB=[math]::Round((($_.Group|Measure-Object Length -Sum).Sum)/1MB,0); Cnt=$_.Count }
  } | Sort-Object MB -Descending | Select-Object -First 10 | ForEach-Object {
    W ("  {0,7} MB  {1,5} 個  {2}" -f $_.MB, $_.Cnt, $_.Ext)
  }
W ""

W "■ Cドライブで容量を食っているフォルダ 上位15（Adobe以外も含む）"
$roots = @('C:\Program Files','C:\Program Files (x86)','C:\ProgramData',
           "$env:LOCALAPPDATA","$env:APPDATA","$env:USERPROFILE",'C:\Windows')
$rows = foreach($r in $roots){
  Get-ChildItem $r -Directory -Force -ErrorAction SilentlyContinue | ForEach-Object {
    $mb = SizeMB $_.FullName
    if($mb -gt 500){ [pscustomobject]@{ MB=$mb; Path=$_.FullName } }
  }
}
$rows | Sort-Object MB -Descending | Select-Object -First 15 | ForEach-Object {
  W ("  {0,7} MB  {1}" -f $_.MB, $_.Path)
}
W ""
W "=== ここまで（何も削除していません）==="

$text = $out -join "`r`n"
try { Set-Clipboard -Value $text; Write-Host "`n（結果はクリップボードにコピー済み。Claudeの入力欄で Ctrl+V）" } catch {}
