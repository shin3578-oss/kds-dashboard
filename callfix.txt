$ErrorActionPreference = 'Stop'
$log = New-Object System.Collections.Generic.List[string]
function Say($m, $c = 'Gray') { Write-Host $m -ForegroundColor $c; $null = $log.Add($m) }
function Step($n, $m) { Say ''; Say "■ [$n] $m" 'Cyan' }
function OK($m)   { Say "   OK   $m" 'Green' }
function NG($m)   { Say "   NG   $m" 'Red' }
function WARN($m) { Say "   注意 $m" 'Yellow' }

         ).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)
if (-not $admin) {
  Write-Host ''
  Write-Host '  管理者として実行されていません。' -ForegroundColor Yellow
  Write-Host '  スタートボタンを右クリック →「ターミナル（管理者）」を開き、同じ1行を貼ってください。' -ForegroundColor Yellow

# --- 1. フォルダとNode.js ---------------------------------------------------
Step 1 'clinic-call フォルダとNode.jsを探す'
$desk = [Environment]::GetFolderPath('Desktop')

# --- 2. IPアドレスの実測 ----------------------------------------------------
Step 2 '受付PCの今のIPアドレスを調べる'
$cfg = Get-NetIPConfiguration | Where-Object {
         $_.IPv4DefaultGateway -and $_.NetAdapter.Status -eq 'Up' -and $_.IPv4Address
       } | Select-Object -First 1
if (-not $cfg) { NG 'ネットワークに繋がっていません。'; return }

# --- 3. 院内CA（rootCA.pem）の設置 -----------------------------------------
Step 3 '院内CAを配信フォルダへ置く（これが「DLできない」の原因）'
$mk = Join-Path $root 'mkcert.exe'
if (-not (Test-Path $mk)) { NG "mkcert.exe がありません: $mk"; return }
Push-Location $root
try {
  $caroot = (& $mk -CAROOT 2>&1 | Select-Object -First 1).ToString().Trim()
  $caPem  = Join-Path $caroot 'rootCA.pem'
  if (-not (Test-Path $caPem)) {
    Say '     院内CAがまだ無いので作成します…'
    & $mk -install 2>&1 | Out-Null
  }
  if (-not (Test-Path $caPem)) { NG "院内CAを作れませんでした: $caPem"; return }
  $pub = Join-Path $root 'public'
  if (-not (Test-Path $pub)) { New-Item -ItemType Directory -Path $pub | Out-Null }
  Copy-Item $caPem (Join-Path $pub 'rootCA.pem') -Force
  OK "配信フォルダへ設置: public\rootCA.pem （元: $caPem）"

  # 秘密鍵が紛れ込んでいないか（絶対に配ってはいけないもの）
  $leak = Join-Path $pub 'rootCA-key.pem'
  if (Test-Path $leak) { Remove-Item $leak -Force; WARN 'public にあった rootCA-key.pem（秘密鍵）を削除しました' }

  # --- 4. 証明書が今のIP向けか ---------------------------------------------
  Step 4 'サーバー証明書が今のIP向けかを確認'
  $certFile = Join-Path $root 'certs\cert.pem'
  $needNew = $true
  if (Test-Path $certFile) {
    try {
      $c = New-Object System.Security.Cryptography.X509Certificates.X509Certificate2($certFile)
      $san = ($c.Extensions | Where-Object { $_.Oid.Value -eq '2.5.29.17' })
      $sanText = if ($san) { $san.Format($false) } else { '' }
      if ($sanText -match [regex]::Escape($ip)) {
        if ($c.NotAfter -gt (Get-Date).AddDays(14)) {
          $needNew = $false
          OK ("今のIPに一致・有効期限 " + $c.NotAfter.ToString('yyyy/MM/dd') + " まで")
        } else { WARN '有効期限が近いので作り直します' }
      } else { WARN "証明書が今のIP($ip)向けではありません。作り直します" }
    } catch { WARN '証明書が読めませんでした。作り直します' }
  } else { WARN '証明書がありません。作ります' }

  if ($needNew) {
    if (-not (Test-Path (Join-Path $root 'certs'))) { New-Item -ItemType Directory -Path (Join-Path $root 'certs') | Out-Null }
    & $mk -key-file 'certs\key.pem' -cert-file 'certs\cert.pem' $ip 'localhost' '127.0.0.1' 2>&1 | Out-Null
    if (Test-Path $certFile) { OK "証明書を $ip 向けに作り直しました（iPadは証明書を入れ直す必要があります）" }
    else { NG '証明書の作成に失敗しました'; return }
  }
} finally { Pop-Location }

function lanIp() {
  for (const list of Object.values(os.networkInterfaces())) {
    for (const n of list || []) {
      if (n.family === 'IPv4' && !n.internal && n.address.startsWith('192.168.')) return n.address;
    }
  }
  return 'この受付PCのIP';
}

http.createServer((req, res) => {
  const p = (req.url || '/').split('?')[0];
  if (p === '/' || p === '/index.html') {
    res.writeHead(200, { 'Content-Type': 'text/html; charset=utf-8', 'Cache-Control': 'no-store' });
    res.end(page());
    return;
  }
  if (p === '/rootCA.pem') {
    fs.readFile(CA, (err, data) => {
      if (err) { res.writeHead(404); res.end('rootCA.pem がありません'); return; }
      // この種別で返さないとiOSが「プロファイル」として取り込まない
      res.writeHead(200, {
        'Content-Type': 'application/x-x509-ca-cert',
        'Content-Disposition': 'attachment; filename="kds-rootCA.pem"',
        'Cache-Control': 'no-store',
      });
      res.end(data);
    });
    return;
  }
  res.writeHead(404); res.end('Not Found');
}).listen(PORT, () => {
  console.log(`[cert-share] http://${lanIp()}:${PORT}/ で証明書の入れ方ページを公開中`);
});

# --- 6. ファイアウォール ----------------------------------------------------
Step 6 '院内から繋がるようにファイアウォールを開ける'
foreach ($p in 3000, 3001) {
  $name = "KDS院内呼び出し TCP$p"
  if (-not (Get-NetFirewallRule -DisplayName $name -ErrorAction SilentlyContinue)) {
    New-NetFirewallRule -DisplayName $name -Direction Inbound -Action Allow `
      -Protocol TCP -LocalPort $p -Profile Any | Out-Null
    OK "$p 番を許可しました"
  } else { OK "$p 番はすでに許可済み" }
}

# --- 7. 起動 & 自動起動 -----------------------------------------------------
Step 7 'サーバーを起動し、PC起動時に自動で立ち上がるようにする'

$taskName = 'KDS-ClinicCall'
try {
  if (Get-ScheduledTask -TaskName $taskName -ErrorAction SilentlyContinue) {
    Unregister-ScheduledTask -TaskName $taskName -Confirm:$false
  }
  $a = New-ScheduledTaskAction -Execute $cmd
  $t = New-ScheduledTaskTrigger -AtLogOn
  $s = New-ScheduledTaskSettingsSet -AllowStartIfOnBatteries -DontStopIfGoingOnBatteries -StartWhenAvailable
  $p = New-ScheduledTaskPrincipal -UserId "$env:USERDOMAIN\$env:USERNAME" -LogonType Interactive -RunLevel Highest
  Register-ScheduledTask -TaskName $taskName -Action $a -Trigger $t -Settings $s -Principal $p | Out-Null
  OK 'PC起動（サインイン）時に自動で立ち上がるようにしました'
} catch { WARN "自動起動の登録に失敗: $($_.Exception.Message)" }

# 既に動いているものは残し、止まっているものだけ起こす
function PortUp($port) {
  try { return [bool](Get-NetTCPConnection -LocalPort $port -State Listen -ErrorAction SilentlyContinue) }
  catch { return $false }
}
if (-not (PortUp 3000)) {
  Start-Process -FilePath $nodeExe -ArgumentList 'server.js' -WorkingDirectory $root -WindowStyle Minimized
  Start-Sleep -Seconds 2
}
if (-not (PortUp 3001)) {
  Start-Process -FilePath $nodeExe -ArgumentList 'cert-share.js' -WorkingDirectory $root -WindowStyle Minimized
  Start-Sleep -Seconds 2
}
if (PortUp 3000) { OK '呼び出しサーバー（3000番）稼働中' } else { NG '呼び出しサーバー（3000番）が起動できませんでした' }
if (PortUp 3001) { OK '証明書ページ（3001番）稼働中' } else { NG '証明書ページ（3001番）が起動できませんでした' }
